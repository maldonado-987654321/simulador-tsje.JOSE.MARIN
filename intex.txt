<script>
let intPartido = "HONOR COLORADO";
let intLista = "LISTA 2";
let intImg = "amalia.png";
let intNombre = "Amalia Vargas";

let junPartido = "HONOR COLORADO";
let junLista = "LISTA 2";

function mostrarPantalla(id){
  let pantallas=document.getElementsByClassName("pantalla");
  for(let i=0;i<pantallas.length;i++){
    pantallas[i].className="pantalla";
  }
  document.getElementById(id).className="pantalla activa";
  window.scrollTo(0,0);
}

function limpiarSeleccion(id){
  let cont=document.getElementById(id);
  let cards=cont.getElementsByClassName("card");
  for(let i=0;i<cards.length;i++){
    cards[i].className=cards[i].className.replace(" seleccionado","");
  }
}

function elegirIntendente(el, partido, lista, img, nombre){
  limpiarSeleccion("p2");
  el.className += " seleccionado";

  intPartido = partido;
  intLista = lista;
  intImg = img;
  intNombre = nombre;

  document.getElementById("finalIntPartido").innerHTML = partido;
  document.getElementById("finalIntLista").innerHTML = lista;
  document.getElementById("finalIntImg").src = img;
  document.getElementById("finalIntNombre").innerHTML = nombre;

  setTimeout(function(){mostrarPantalla("p3")},120);
}

function elegirLista(el, partido, lista){
  limpiarSeleccion("p3");
  el.className += " seleccionado";

  junPartido = partido;
  junLista = lista;

  document.getElementById("tituloListaJunta").innerHTML = lista;
  document.getElementById("tituloPartidoJunta").innerHTML = partido;
  document.getElementById("finalJunPartido").innerHTML = partido;
  document.getElementById("finalJunLista").innerHTML = lista;

  setTimeout(function(){mostrarPantalla("p6")},120);
}

function elegirJunta(el,img,nombre,opcion){
  limpiarSeleccion("p6");
  el.className += " seleccionado";

  document.getElementById("finalJunImg").src = img;
  document.getElementById("finalJunNombre").innerHTML = nombre;
  document.getElementById("finalJunOpcion").innerHTML = opcion;

  setTimeout(function(){mostrarPantalla("p7")},120);
}

function volverA(id){
  mostrarPantalla(id);
}

function reiniciar(){
  mostrarPantalla("p2");
}

window.onload=function(){
  mostrarPantalla("p2");
}
